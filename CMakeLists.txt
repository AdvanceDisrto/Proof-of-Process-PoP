cmake_minimum_required(VERSION 3.20)
project(iSPU_Sovereign LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_library(spu_reference src/spu_reference.cpp)
target_include_directories(spu_reference PUBLIC include)
target_compile_options(spu_reference PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(spu_reference_tests tests/test_spu.cpp)
target_link_libraries(spu_reference_tests PRIVATE spu_reference)
add_test(NAME Reference_Vector_Math COMMAND spu_reference_tests vector)
add_test(NAME SECDED_ECC_Logic COMMAND spu_reference_tests ecc)
add_test(NAME MMIO_Host_Boundary COMMAND spu_reference_tests mmio)
add_test(NAME Materialization_Fail_Closed COMMAND Python3::Interpreter ${CMAKE_SOURCE_DIR}/tools/validate_materialization_topology.py ${CMAKE_SOURCE_DIR}/materialization/topology.pending.json --expect-blocked)
